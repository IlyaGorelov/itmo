org 0
array_i: word 0x12C ; счетчик для начального массива
len: word 6; размер нач массива
odd_array_i: word 0x140; счетчик для массива нечетных
count_odds: word 0; 
even_array_i: word 0x154;  счетчик для массива четных
count_evens: word 0


result_odd_1: word 0; младшая часть результата для нечетных
result_odd_2: word 0; старшая часть результата для нечетных

result_even_1: word 0; младшая часть результата для четных
result_even_2: word 0; старшая часть результата для четных

max_1: word 0; мл часть резульата  ячейка A
max_2: word 0; ст часть результата ячейка B

temp_1: word 0; служит чтобы хранить младшую часть полученного числа из массива

start: 
;-----------Делим массив на массив четных и нечетных
cla
get_el: ld (array_i)+; получаем элемент-младшую часть
st temp_1; сохраняем во временную переменную
ror; передаем в CF правый бит
bcc add_even ; если CF=0 переходим на добавление в массив четных
ld temp_1; получаем неизмененную младшую часть
st (odd_array_i)+; иначе set в массив нечетных младшей части элемента
ld (array_i)+; получаем элемент-старшую часть
st (odd_array_i)+; set в массив нечетных старшей части элемента

ld count_odds; получаем счетчик нечетных
inc; +1
st count_odds; сохраняем его
jump loop1; переходим на loop

add_even: ld temp_1
st (even_array_i)+ ; сохраняем младшую часть
ld (array_i)+
st (even_array_i)+; и старшую
ld count_evens
inc
st count_evens

loop1:loop len
jump get_el

;-----------------считаем сумму нечетных
cla
 
get_odds_el: ld -(odd_array_i); получили старшую часть
push; в стеке 9FF-старшая часть числа 1
ld -(odd_array_i); получили младшую часть
push; в стеке 9FE-младшая часть числа 1

ld result_odd_2; получили старшую часть
push; в стеке 9FD-старшая часть числа 2
ld result_odd_1; получили младшую часть
push; в стеке 9FC-младшая часть числа 2

call $sum; в стеке 9FB - служебный IP
;стек указывает на 9FC
pop;9FD
pop;9FE

pop; получили младшую часть
st result_odd_1
pop
st result_odd_2

loop count_odds
jump get_odds_el

;-----------------считаем сумму четных
cla
 
get_evens_el: ld -(even_array_i); получили старшую часть
push; в стеке 9FF-старшая часть числа 1
ld -(even_array_i); получили младшую часть
push; в стеке 9FE-младшая часть числа 1

ld result_even_2; получили старшую часть
push; в стеке 9FD-старшая часть числа 2
ld result_even_1; получили младшую часть
push; в стеке 9FC-младшая часть числа 2

call $sum; в стеке 9FB - служебный IP
;стек указывает на 9FC
pop;9FD
pop;9FE

pop; получили младшую часть
st result_even_1
pop
st result_even_2

loop count_evens
jump get_evens_el

;----------------------------------сравним их
ld result_odd_1
push; 9FF - младшая часть 1
ld result_odd_2
push; 9FE - старшая часть 1

ld result_even_1
push; 9FD - младшая часть 2
ld result_even_2
push; 9FC - старшая часть 2

call $compare
pop
pop
pop
pop; результат сравнения
bns odd_max; переход если сумма четных по модулю меньше 
;-------если сумма четных больше или равна то сохр в результат ее
hlt
ld result_even_1
st max_1
ld result_even_2
st max_2
jump e

odd_max:
ld result_odd_1
st max_1
ld result_odd_2
st max_2

e: hlt

org 300;array
e1_1: word 0x2ea
e1_2: word 0x56

e2_1: word 0x1
e2_2: word 0x6

e3_1: word 0x1
e3_2: word 0x6

e4_1: word 0x2ea
e4_2: word 0x2eb

e5_1: word 0x2ea
e5_2: word 0x2eb

e6_1: word 0x2ea
e6_2: word 0x2eb

;odd array starts from 70
;even array from 90

org 500
sum:
ld &1; получили младшую часть 1
add &3; добавили младшую часть 2
st &3; сохранили на 9FE 
ld &2
adc &4
st &4; сохранили в 9FF
ret

org 550
compare:
;-----посчитаем модуль числа 1
ld &2; получить мл часть чет
st even_1
ld &1; получить ст часть четного
st even_2
cmp #0;
bpl get_abs_for_odd; если >0  то переход
not
st even_2
ld &2; получить мл часть четного
st even_1
neg
st even_1
ld even_2
adc #0
st even_2

;-----посчитаем модуль числа 2
get_abs_for_odd: ld &4; получить мл часть нечет
st odd_1
ld &3; получить ст часть нечетного
st odd_2
cmp #0;
bpl invert
not
st odd_2
ld &4; получить мл часть нечетного
st odd_1
neg
st odd_1
ld odd_2
adc #0
st odd_2

;будем вычитать из четного нечетное, инвертируем нечетное
invert: ld odd_2
not
st odd_2
ld odd_1
neg
st odd_1
cla
adc odd_2
st odd_2
;вычтем
ld odd_2; получили ст часть нечет
push
ld odd_1; получили мл часть нечет
push
ld even_2; получили ст часть чет
push
ld even_1; получили мл часть чет
push

call $sum
pop
pop
pop; получили младшую часть
pop; и старшую
cmp #0
bzs zero; если AC=0 переход
bns smaller; AC<0
bnc bigger; ac>0

zero: cla
jump set
bigger: ld #1
jump set
smaller: ld#-1
set: st &4

ret
odd_1: word 0
odd_2: word 0
even_1: word 0
even_2: word 0
