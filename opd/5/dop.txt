start:
spin1: in 0x19
and #0x40
beq spin1
in 0x18
st temp_sym

cmp #0x0a; если символ соответствует переносу строки
beq continue; то заканчиваем луп

push
call $check_sym
pop

cmp #1;
bne spin1; если неверно, то ждем нового символа
ld temp_sym; иначе берем символ и сохраняем
st (cur_sym)+
ld len
inc
st len
jump spin1
;иначе продолжаем программу

;====проверим длину слова===
continue: ld len
push
call $check_len
pop
cmp #1
bne call_wrong
;======переведем число и месяц в цифровые значения
ld first_sym
st cur_sym
ld (cur_sym)+
push; в 9ff - d1
ld (cur_sym)+
push; в 9fe - d2
call $to_int
pop
pop
st day

ld (cur_sym)+;пропустим точку
ld (cur_sym)+
push; в 9ff - m1
ld (cur_sym)+
push; в 9fe - m2
call $to_int
pop
pop
st month
;=====проверим числа=========
ld first_sym
add #2; добавим и получим адрес точки
st cur_sym

ld (cur_sym)    
cmp #0x2E
bne call_wrong; проверили .

ld month
push
ld max_m
push 
call $check_is_in_borders
pop
pop
cmp #1
bne call_wrong; проверили месяц


ld month
cmp #2
bne other_months
ld day; проверим то ли количество дней, если месяц февраль (28)
push
ld max_d_in_feb
push
call $check_is_in_borders
pop
pop
cmp #1
bne call_wrong; проверили дни
jump output
other_months:
ror
bhis odd_months
ld day; проверим то ли количество дней, если месяц четный (30)
push
ld max_d_in_even
push
call $check_is_in_borders
pop
pop
cmp #1
bne call_wrong; проверили дни
jump output
odd_months:
ld day; проверим то ли количество дней, если месяц нечетный (31)
push
ld max_d_in_odd
push
call $check_is_in_borders
pop
pop
cmp #1
bne call_wrong; проверили дни
jump output
call_wrong:
call $wrong

check_sym:
ld &1
cmp #0x2E; сравниваю с .
beq R2; если . , то ret 1
cmp #0x30; сравниваю с 0
blt R1; если < кода 0, то ret 0
cmp #0x3A; сравниваю с 9
bge R1; если > 9, то ret 0
jump R2; иначе ret 1
R1:
ld #0
st &1
ret
R2: 
ld #1
st &1
ret

check_len:
ld &1
cmp #5
bne R3
ld #1
st &1
ret
R3:
ld #0
st &1
ret
